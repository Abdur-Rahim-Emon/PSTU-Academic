4. division 
.model small
.stack 100h
.data
    a db "Input first number: $"
    b db "Input second number: $"
    n db 13,10,"$"
    q db 13,10,"Quotient  = $"
    r db 13,10,"Remainder = $"
x db ?
y db ?

.code
main proc
    mov ax, @data
    mov ds, ax

    ; Input first number (dividend)
    mov ah, 9
    lea dx, a
    int 21h

    mov ah, 1
    int 21h
    sub al, 48
    mov x, al

    mov ah, 9
    lea dx, n
    int 21h

    ; Input second number (divisor)
    mov ah, 9
    lea dx, b
    int 21h

    mov ah, 1
    int 21h
    sub al, 48
    mov y, al

    mov ah, 9
    lea dx, n
    int 21h

    ; -- DIVISION --
    mov al, x
    mov ah, 0     ; ? clear AH (very important!)
    div y         ; AL = quotient, AH = remainder

    mov bl, al    ; save quotient
    mov bh, ah    ; save remainder

    ; Print quotient
    mov ah, 9
    lea dx, q
    int 21h

    mov ah, 2
    mov dl, bl
    add dl, 48    ; ? number to ASCII
    int 21h

    ; Print remainder
    mov ah, 9
    lea dx, r
    int 21h

    mov ah, 2
    mov dl, bh
    add dl, 48    ; ? number to ASCII
    int 21h

    mov ah, 4ch
    int 21h
main endp
end main

    mov x, al

    ;---------------------------
    ; INPUT SECOND NUMBER
    ;---------------------------
    mov ah, 9
    lea dx, msg2
    int 21h

    mov ah, 1
    int 21h
    sub al, 48
    mov y, al

    ;---------------------------
    ; INPUT THIRD NUMBER
    ;---------------------------
    mov ah, 9

    int 21h

    mov z, al

    ;---------------------------
    ; CALCULATE AVERAGE
    ; avg = (x + y + z) / 3
    ;---------------------------
    mov al, x

    add al, z         ; AL = x + y + z

    mov ch, ah        ; save remainder in CH

    ;---------------------------
    ; PRINT RESULT
    ;---------------------------
    mov ah, 9
    lea dx, msg4
    int 21h

    ; print average digit

    mov dl, cl
    add dl, 48        ; number ? ASCII
    int 21h

    ; optional: print remainder (decimal part hint)
    ; uncomment below if you want to show remainder
    ; mov ah, 2
    ; mov dl, ch
    ; add dl, 48
    ; int 21h

    ;---------------------------
    ; EXIT
    ;---------------------------
    mov ah, 4ch
    int 21h

main endp

Mid solve
.model small
.stack 100h

.data
    menu_msg    db 13, 10, "=========================="
                db 13, 10, "       CALCULATOR         "
                db 13, 10, "=========================="
                db 13, 10, "1. Addition"
                db 13, 10, "2. Subtraction"
                db 13, 10, "3. Multiplication"
                db 13, 10, "4. Division"
                db 13, 10, "5. Exit"
                db 13, 10, "=========================="
                db 13, 10, "Choose Operation (1-5): $"
    
    msg_num1    db 13, 10, "Enter First Number (0-9): $"
    msg_num2    db 13, 10, "Enter Second Number (0-9): $"
    msg_res     db 13, 10, "Result: $"
    msg_div_err db 13, 10, "Error: Division by zero$"
    msg_cont    db 13, 10, "Continue? (1/0): $"

    num1        db ?
    num2        db ?
    choice      db ?

.code
main proc
    mov ax, @data
    mov ds, ax

program_loop:
    ; 1. Display Menu
    mov ah, 09h
    lea dx, menu_msg
    int 21h

    ; Input: Choice
    mov ah, 01h
    int 21h
    sub al, '0'        ; Convert ASCII to Integer
    mov choice, al

    ; Exit condition check
    cmp choice, 5
    je exit_program

    ; Validate operation choice (1-4)
    cmp choice, 1
    jl program_loop
    cmp choice, 4
    jg program_loop

    ; 2. Input: First Number
    mov ah, 09h
    lea dx, msg_num1
    int 21h
    
    mov ah, 01h
    int 21h
    sub al, '0'
    mov num1, al

    ; Input: Second Number
    mov ah, 09h
    lea dx, msg_num2
    int 21h
    
    mov ah, 01h
    int 21h
    sub al, '0'
    mov num2, al

    ; 3. Branching based on operation
    cmp choice, 1
    je do_addition
    cmp choice, 2
    je do_subtraction
    cmp choice, 3
    je do_multiplication
    cmp choice, 4
    je do_division

; --- ADDITION ---
do_addition:
    mov al, num1
    add al, num2
    mov ah, 0
    jmp print_result

; --- SUBTRACTION ---
do_subtraction:
    mov al, num1
    cmp al, num2
    jl sub_negative     ; Handle negative results (A < B)

    sub al, num2
    mov ah, 0
    jmp print_result

sub_negative:
    mov ah, 09h
    lea dx, msg_res
    int 21h

    ; Print negative sign '-'
    mov dl, '-'
    mov ah, 02h
    int 21h

    mov al, num2
    sub al, num1
    mov ah, 0
    jmp display_digits

; --- MULTIPLICATION ---
do_multiplication:
    mov al, num1
    mov bl, num2
    mul bl              ; AX = AL * BL
    jmp print_result

; --- DIVISION ---
do_division:
    ; Check for Division by Zero
    cmp num2, 0
    je div_by_zero

    mov al, num1
    mov ah, 0
    mov bl, num2
    div bl              ; AL = Quotient, AH = Remainder
    mov ah, 0           ; Keep quotient in AX
    jmp print_result

div_by_zero:
    mov ah, 09h
    lea dx, msg_div_err
    int 21h
    jmp ask_continue

; --- PRINT RESULT ROUTINE ---
print_result:
    push ax             ; Save AX
    mov ah, 09h
    lea dx, msg_res
    int 21h
    pop ax              ; Restore AX

display_digits:
    aam                 ; AH = Tens digit, AL = Units digit
    mov bx, ax          ; Save result in BX (BH = Tens, BL = Units)

    ; Print tens digit if > 0
    cmp bh, 0
    je print_units

    mov dl, bh
    add dl, '0'
    mov ah, 02h
    int 21h

print_units:
    mov dl, bl
    add dl, '0'
    mov ah, 02h
    int 21h

; --- LOOP / CONTINUE CHECK ---
ask_continue:
    mov ah, 09h
    lea dx, msg_cont
    int 21h

    mov ah, 01h
    int 21h

    cmp al, '1'
    je program_loop     ; Repeat if user inputs '1'

exit_program:
    mov ah, 4ch
    int 21h
main endp
end main
