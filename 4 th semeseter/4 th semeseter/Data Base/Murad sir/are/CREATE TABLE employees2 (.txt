CREATE TABLE employees2 (
      emp_id      NUMBER PRIMARY KEY,
        emp_name    VARCHAR2(50),
        salary      NUMBER
   );

CREATE OR REPLACE TRIGGER trg_check_salary2
    BEFORE INSERT OR UPDATE ON employees2
    FOR EACH ROW
    BEGIN
        -- Check salary condition
       IF :NEW.salary > 100000 THEN
            -- Raise custom error with error code and message
            RAISE_APPLICATION_ERROR(-20001, 'Salary cannot exceed 100,000.');
        END IF;
   END;
  /

CREATE OR REPLACE TRIGGER trg_validate_employee7
    BEFORE INSERT OR UPDATE ON employees7
    FOR EACH ROW
    DECLARE
        PRAGMA AUTONOMOUS_TRANSACTION;  -- ✅ Allows separate commit
        v_error_msg VARCHAR2(200);
    BEGIN
        -- Salary validation
        IF :NEW.salary < 10000 OR :NEW.salary > 100000 THEN
           v_error_msg := 'Salary must be between 10,000 and 100,000.';

           INSERT INTO error_log7 (error_time, emp_id, error_msg)
           VALUES (SYSTIMESTAMP, :NEW.emp_id, v_error_msg);

          COMMIT; -- ✅ Commit independent transaction
           RAISE_APPLICATION_ERROR(-20002, v_error_msg);
       END IF;
 
     -- Department validation
      IF :NEW.department NOT IN ('HR', 'IT', 'SALES') THEN
           v_error_msg := 'Department must be HR, IT, or SALES.';

           INSERT INTO error_log7 (error_time, emp_id, error_msg)
          VALUES (SYSTIMESTAMP, :NEW.emp_id, v_error_msg);

          COMMIT; -- ✅ Commit independent transaction
 
          RAISE_APPLICATION_ERROR(-20003, v_error_msg);
       END IF;
   END;
  /
